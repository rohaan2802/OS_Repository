#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>

int main() {
    pid_t childpid = fork();

    if (childpid == 0) { // Child process
        printf("I am a child process with PID = %d\n", getpid());
        printf("Now executing 'ls -l' using execl()\n");

        execl("/bin/ls", "ls", "-l", NULL);

        // If execl fails, the below line will execute
        printf("Execl failed\n");
    } else if (childpid > 0) { // Parent process
        wait(NULL);
        printf("I am parent process with PID = %d, child finished.\n", getpid());
    }

    return 0;
}
