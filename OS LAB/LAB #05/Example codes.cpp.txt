/* FILE DESCRIPTOR EXAMPLE */
#include <iostream>
#include <unistd.h>
using namespace std;

int main() {
    char buffer[10];
    cout << "Enter string: ";
    read(0, buffer, 10);  // Read from standard input (keyboard)
    write(1, buffer, 10); // Write to standard output (monitor)
    return 0;
}

/* PIPE EXAMPLE 1: Parent writes, child reads */
#include <iostream>
#include <unistd.h>
using namespace std;

int main() {
    int fd[2];
    pipe(fd); // Create pipe
    pid_t pid = fork();
    
    if (pid > 0) { // Parent process
        close(fd[0]); // Close read end
        char msg[] = "Hello Child";
        write(fd[1], msg, sizeof(msg));
        close(fd[1]); // Close write end
    } else { // Child process
        close(fd[1]); // Close write end
        char buffer[20];
        read(fd[0], buffer, sizeof(buffer));
        cout << "Child received: " << buffer << endl;
        close(fd[0]); // Close read end
    }
    return 0;
}

/* PIPE EXAMPLE 2: Bi-directional communication */
#include <iostream>
#include <unistd.h>
using namespace std;

int main() {
    int fd1[2], fd2[2];
    pipe(fd1); // Pipe 1: Parent -> Child
    pipe(fd2); // Pipe 2: Child -> Parent
    pid_t pid = fork();
    
    if (pid > 0) { // Parent process
        close(fd1[0]); // Close read end of Pipe 1
        close(fd2[1]); // Close write end of Pipe 2
        
        char msg1[] = "Hello Child";
        write(fd1[1], msg1, sizeof(msg1));
        close(fd1[1]);
        
        char buffer[20];
        read(fd2[0], buffer, sizeof(buffer));
        cout << "Parent received: " << buffer << endl;
        close(fd2[0]);
    } else { // Child process
        close(fd1[1]); // Close write end of Pipe 1
        close(fd2[0]); // Close read end of Pipe 2
        
        char buffer[20];
        read(fd1[0], buffer, sizeof(buffer));
        cout << "Child received: " << buffer << endl;
        close(fd1[0]);
        
        char msg2[] = "Hello Parent";
        write(fd2[1], msg2, sizeof(msg2));
        close(fd2[1]);
    }
    return 0;
}
